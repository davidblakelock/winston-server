// ─── Types ────────────────────────────────────────────────────────────────
interface Bill {
  id: string;
  name: string;
  dueDay: number;
  amount?: string;
  category?: string;
  frequency?: string;
  notes?: string;
  isPaid: boolean;
  isOverdue: boolean;
  daysUntilDue: number;
  nextDueDateLabel: string;
  paidThroughDate?: string;
  lastRemindedDate?: string;
}

// ─── Helpers ──────────────────────────────────────────────────────────────
const getDaysUntilDue = (dueDay: number): number => {
  const today = new Date();
  const currentDay = today.getDate();
  const daysInMonth = new Date(today.getFullYear(), today.getMonth() + 1, 0).getDate();

  if (dueDay >= currentDay) {
    return dueDay - currentDay;
  } else {
    return (daysInMonth - currentDay) + dueDay;
  }
};

const getDueStatus = (bill: Bill): { label: string; color: string } => {
  if (bill.isPaid) return { label: 'Paid ✓', color: COLORS.green };
  if (bill.isOverdue) return { label: 'Overdue', color: COLORS.danger };
  if (bill.daysUntilDue <= 3) return { label: `Due in ${bill.daysUntilDue} day${bill.daysUntilDue === 1 ? '' : 's'}`, color: COLORS.danger };
  if (bill.daysUntilDue <= 7) return { label: `Due in ${bill.daysUntilDue} days`, color: COLORS.yellow };
  return { label: `Due ${bill.nextDueDateLabel}`, color: COLORS.muted };
};

const getOrdinal = (n: number): string => {
  const s = ['th', 'st', 'nd', 'rd'];
  const v = n % 100;
  return s[(v - 20) % 10] || s[v] || s[0];
};

const formatAmount = (amount?: number): string => {
  if (!amount) return '';
  return `$${amount.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
};

const formatLastPaid = (date?: string): string => {
  if (!date) return 'Never paid';
  return `Last paid ${new Date(date).toLocaleDateString([], { month: 'short', day: 'numeric' })}`;
};

// ─── Add/Edit Modal ───────────────────────────────────────────────────────
const BillFormModal = ({ visible, bill, onSave, onClose }: {
  visible: boolean;
  bill?: Bill;
  onSave: (data: Partial<Bill>) => void;
  onClose: () => void;
}) => {

  const [billerName, setBillerName] = useState(bill?.name || '');
  const [duéDay, setDueDay] = useState(bill?.dueDay?.toString() || '');
  const [amount, setAmount] = useState(bill?.amount || '');
  const [notes, setNotes] = useState(bill?.notes || '');

  React.useEffect(() => {
    if (visible && bill) {
      setBillerName(bill.name || '');
      setDueDay(bill.dueDay?.toString() || '');
      setAmount(bill.amount || '');
      setNotes(bill.notes || '');
    } else if (visible && !bill) {
      setBillerName('');
      setDueDay('');
      setAmount('');
      setNotes('');
    }
  }, [visible]);

  const handleSave = () => {
    if (!billerName.trim() || !duéDay.trim()) return;
    const day = parseInt(duéDay);
    if (isNaN(day) || day < 1 || day > 31) return;

      onSave({
      name: billerName.trim(),
      dueDay: day,
      amount: amount ? amount : undefined,
      category: 'other',
      frequency: 'monthly',
      notes: notes.trim() || undefined,
    });


  };

  const isValid = billerName.trim() && duéDay.trim() &&
    !isNaN(parseInt(duéDay)) && parseInt(duéDay) >= 1 && parseInt(duéDay) <= 31;

  return (
    <Modal visible={visible} animationType="slide" presentationStyle="pageSheet">
      <SafeAreaView style={styles.safe}>
        <View style={{ height: 40 }} />
        <View style={styles.modalHeader}>
          <TouchableOpacity onPress={onClose}>
            <Text style={styles.modalCancel}>Cancel</Text>
          </TouchableOpacity>
          <Text style={styles.modalTitle}>{bill ? 'Edit Bill' : 'Add Bill'}</Text>
          <TouchableOpacity onPress={handleSave} disabled={!isValid}>
            <Text style={[styles.modalSave, !isValid && { opacity: 0.4 }]}>Save</Text>
          </TouchableOpacity>
        </View>

        <KeyboardAvoidingView
          style={{ flex: 1 }}
          behavior={Platform.OS === 'ios' ? 'padding' : undefined}
        >
          <ScrollView style={styles.scroll} showsVerticalScrollIndicator={false}>

            <View style={styles.formSection}>
              <Text style={styles.formLabel}>PAYEE NAME</Text>
              <TextInput
                style={styles.formInput}
                value={billerName}
                onChangeText={setBillerName}
                placeholder="e.g. Visa Credit Card, Water Bill"
                placeholderTextColor={COLORS.muted}
                autoFocus
              />
            </View>

            <View style={styles.formSection}>
              <Text style={styles.formLabel}>DUE DAY OF MONTH</Text>
              <TextInput
                style={styles.formInput}
                value={duéDay}
                onChangeText={setDueDay}
                placeholder="e.g. 15"
                placeholderTextColor={COLORS.muted}
                keyboardType="number-pad"
                maxLength={2}
              />
              <Text style={styles.formHint}>Enter the day of month this bill is typically due (1-31)</Text>
            </View>

            <View style={styles.formSection}>
              <Text style={styles.formLabel}>TYPICAL AMOUNT (OPTIONAL)</Text>
              <TextInput
                style={styles.formInput}
                value={amount}
                onChangeText={setAmount}
                placeholder="e.g. 125.00"
                placeholderTextColor={COLORS.muted}
                keyboardType="decimal-pad"
              />
            </View>

            <View style={styles.formSection}>
              <Text style={styles.formLabel}>NOTES (OPTIONAL)</Text>
              <TextInput
                style={[styles.formInput, styles.formInputMulti]}
                value={notes}
                onChangeText={setNotes}
                placeholder="e.g. Pay via Venmo to John Smith, login at chase.com"
                placeholderTextColor={COLORS.muted}
                multiline
                numberOfLines={3}
              />
            </View>

            <View style={{ height: 40 }} />
          </ScrollView>
        </KeyboardAvoidingView>
      </SafeAreaView>
    </Modal>
  );
};

// ─── Main Screen ──────────────────────────────────────────────────────────
export default function BillsScreen() {
  const router = useRouter();
  const [bills, setBills] = useState<Bill[]>([]);
  const [isLoading, setIsLoading] = useState(true);
  const [selectedBill, setSelectedBill] = useState<Bill | null>(null);
  const [showDetail, setShowDetail] = useState(false);
  const [showForm, setShowForm] = useState(false);
  const [editingBill, setEditingBill] = useState<Bill | undefined>(undefined);

  const fetchBills = useCallback(async () => {
    try {
      const res = await fetch(`${SERVER}/api/bills`, {
        headers: { 'x-api-key': API_KEY },
      });
      const data = await res.json();
      const billList = Array.isArray(data.bills) ? data.bills : Array.isArray(data) ? data : [];
      // Sort by days until due

      billList.sort((a: Bill, b: Bill) => a.daysUntilDue - b.daysUntilDue);
      setBills(billList);

    } catch (err) {
      console.error('Failed to fetch bills:', err);
    } finally {
      setIsLoading(false);
    }
  }, []);

  useFocusEffect(useCallback(() => { fetchBills(); }, [fetchBills]));

  const handleSave = async (data: Partial<Bill>) => {
    try {
      if (editingBill) {
        await fetch(`${SERVER}/api/bills/${editingBill.id}`, {
          method: 'PUT',
          headers: { 'Content-Type': 'application/json', 'x-api-key': API_KEY },
          body: JSON.stringify(data),
        });
      } else {
        await fetch(`${SERVER}/api/bills`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json', 'x-api-key': API_KEY },
          body: JSON.stringify(data),
        });
      }
      setShowForm(false);
      setShowDetail(false);
      setEditingBill(undefined);
      setSelectedBill(null);
      await fetchBills();
    } catch (err) {
      console.error('Save failed:', err);
    }
  };

  const handleMarkPaid = async (bill: Bill) => {
    try {
      await fetch(`${SERVER}/api/bills/${bill.id}/paid`, {
        method: 'POST',
        headers: { 'x-api-key': API_KEY },
      });
      setShowDetail(false);
      setSelectedBill(null);
      await fetchBills();
    } catch (err) {
      console.error('Mark paid failed:', err);
    }
  };

  const handleDelete = (bill: Bill) => {
    Alert.alert('Remove Bill', `Remove ${bill.name}?`, [
      { text: 'Cancel', style: 'cancel' },
      {
        text: 'Remove', style: 'destructive',
        onPress: async () => {
          await fetch(`${SERVER}/api/bills/${bill.id}`, {
            method: 'DELETE',
            headers: { 'x-api-key': API_KEY },
          });
          setShowDetail(false);
          setSelectedBill(null);
          await fetchBills();
        },
      },
    ]);
  };

  // Separate into due soon and upcoming

    const dueSoon = bills.filter(b => b.daysUntilDue <= 7 && !b.isPaid);
  const upcoming = bills.filter(b => b.daysUntilDue > 7 && !b.isPaid);

  return (
    <SafeAreaView style={styles.safe}>
      <StatusBar barStyle="light-content" backgroundColor={COLORS.background} />

      <View style={styles.header}>
        <TouchableOpacity onPress={() => router.back()} style={styles.headerBtn}>
          <Ionicons name="arrow-back" size={22} color={COLORS.gold} />
        </TouchableOpacity>
        <Text style={styles.headerTitle}>Bills</Text>
        <TouchableOpacity
          style={styles.headerBtn}
          onPress={() => { setEditingBill(undefined); setShowForm(true); }}
        >
          <Ionicons name="add" size={24} color={COLORS.gold} />
        </TouchableOpacity>
      </View>
      <View style={styles.divider} />

      {isLoading ? (
        <View style={styles.centered}>
          <ActivityIndicator size="large" color={COLORS.gold} />
        </View>
      ) : bills.length === 0 ? (
        <View style={styles.centered}>
          <Ionicons name="receipt-outline" size={52} color={COLORS.muted} />
          <Text style={styles.emptyTitle}>No Bills Tracked</Text>
          <Text style={styles.emptySubtitle}>
            Add bills that require manual payment. Winston will remind you 3 days before each one is due.
          </Text>
          <TouchableOpacity
            style={styles.addButton}
            onPress={() => { setEditingBill(undefined); setShowForm(true); }}
          >
            <Text style={styles.addButtonText}>Add First Bill</Text>
          </TouchableOpacity>
        </View>
      ) : (
        <ScrollView style={styles.scroll} showsVerticalScrollIndicator={false}>
          {dueSoon.length > 0 && (
            <View style={styles.section}>
              <Text style={styles.sectionTitle}>DUE SOON</Text>
              <View style={styles.billsCard}>
                {dueSoon.map((bill, i) => (
                  <View key={bill.id}>
                    <BillRow
                      bill={bill}
                      onPress={() => { setSelectedBill(bill); setShowDetail(true); }}
                      onMarkPaid={() => handleMarkPaid(bill)}
                    />
                    {i < dueSoon.length - 1 && <View style={styles.rowDivider} />}
                  </View>
                ))}
              </View>
            </View>
          )}

          {upcoming.length > 0 && (
            <View style={styles.section}>
              <Text style={styles.sectionTitle}>UPCOMING</Text>
              <View style={styles.billsCard}>
                {upcoming.map((bill, i) => (
                  <View key={bill.id}>
                    <BillRow
                      bill={bill}
                      onPress={() => { setSelectedBill(bill); setShowDetail(true); }}
                      onMarkPaid={() => handleMarkPaid(bill)}
                    />
                    {i < upcoming.length - 1 && <View style={styles.rowDivider} />}
                  </View>
                ))}
              </View>
            </View>
          )}

          <Text style={styles.reminderNote}>
            Winston will remind you 3 days before each bill is due.
          </Text>

          <View style={{ height: 40 }} />
        </ScrollView>
      )}

      {/* Detail modal */}
      <BillDetailModal
        bill={selectedBill}
        visible={showDetail}
        onClose={() => { setShowDetail(false); setSelectedBill(null); }}
        onEdit={() => { setShowDetail(false); setEditingBill(selectedBill!); setShowForm(true); }}
        onDelete={() => handleDelete(selectedBill!)}
        onMarkPaid={() => handleMarkPaid(selectedBill!)}
      />

      {/* Add/Edit form */}
      <BillFormModal
        visible={showForm}
        bill={editingBill}
        onSave={handleSave}
        onClose={() => { setShowForm(false); setEditingBill(undefined); }}
      />
    </SafeAreaView>
  );
}
