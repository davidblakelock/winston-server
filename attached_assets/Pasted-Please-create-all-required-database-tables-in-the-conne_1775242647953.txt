Please create all required database tables in the connected database. Run the following SQL to create each table if it does not already exist.
CREATE TABLE IF NOT EXISTS users (id SERIAL PRIMARY KEY, google_id VARCHAR UNIQUE NOT NULL, email VARCHAR, name VARCHAR, companion_name VARCHAR DEFAULT 'Emma Peel', voice_id VARCHAR, created_at TIMESTAMP DEFAULT NOW(), updated_at TIMESTAMP DEFAULT NOW());
CREATE TABLE IF NOT EXISTS profiles (id SERIAL PRIMARY KEY, user_id INTEGER REFERENCES users(id), home_address VARCHAR, wake_time VARCHAR, timezone VARCHAR DEFAULT 'America/Chicago', preferences JSONB, places JSONB, interests JSONB, health JSONB, created_at TIMESTAMP DEFAULT NOW(), updated_at TIMESTAMP DEFAULT NOW());
CREATE TABLE IF NOT EXISTS reminders (id SERIAL PRIMARY KEY, user_id INTEGER REFERENCES users(id), reminder_text VARCHAR NOT NULL, scheduled_time TIMESTAMP WITH TIME ZONE, status VARCHAR DEFAULT 'pending', is_recurring BOOLEAN DEFAULT FALSE, recurrence_pattern VARCHAR, created_at TIMESTAMP DEFAULT NOW(), updated_at TIMESTAMP DEFAULT NOW());
CREATE TABLE IF NOT EXISTS push_subscriptions (id SERIAL PRIMARY KEY, user_id INTEGER REFERENCES users(id), subscription_json TEXT NOT NULL, device_name VARCHAR, created_at TIMESTAMP DEFAULT NOW());
CREATE TABLE IF NOT EXISTS stories (id SERIAL PRIMARY KEY, user_id INTEGER REFERENCES users(id), prompt_question TEXT, story_text TEXT, created_at TIMESTAMP DEFAULT NOW());
CREATE TABLE IF NOT EXISTS journal_entries (id SERIAL PRIMARY KEY, user_id INTEGER REFERENCES users(id), entry_text TEXT, mood_score INTEGER, created_at TIMESTAMP DEFAULT NOW());
CREATE TABLE IF NOT EXISTS lists (id SERIAL PRIMARY KEY, user_id INTEGER REFERENCES users(id), list_name VARCHAR, items JSONB, created_at TIMESTAMP DEFAULT NOW(), updated_at TIMESTAMP DEFAULT NOW());
CREATE TABLE IF NOT EXISTS conversation_memory (id SERIAL PRIMARY KEY, user_id INTEGER REFERENCES users(id), memory_summary TEXT, conversation_date DATE, created_at TIMESTAMP DEFAULT NOW());
CREATE TABLE IF NOT EXISTS important_dates (id SERIAL PRIMARY KEY, user_id INTEGER REFERENCES users(id), person_name VARCHAR, date_type VARCHAR, date VARCHAR, created_at TIMESTAMP DEFAULT NOW());
CREATE TABLE IF NOT EXISTS bills (id SERIAL PRIMARY KEY, user_id INTEGER REFERENCES users(id), bill_name VARCHAR, due_day INTEGER, frequency VARCHAR, amount DECIMAL, reminder_days_before INTEGER DEFAULT 3, created_at TIMESTAMP DEFAULT NOW());
CREATE TABLE IF NOT EXISTS recommendations (id SERIAL PRIMARY KEY, user_id INTEGER REFERENCES users(id), recommendation_type VARCHAR, recommendation_text TEXT, recommended_at TIMESTAMP DEFAULT NOW(), followed_up BOOLEAN DEFAULT FALSE, follow_up_response TEXT);
CREATE TABLE IF NOT EXISTS shows (id SERIAL PRIMARY KEY, user_id INTEGER REFERENCES users(id), show_name VARCHAR, status VARCHAR DEFAULT 'watching', created_at TIMESTAMP DEFAULT NOW());
